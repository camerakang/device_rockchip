cmake_minimum_required(VERSION 3.20)
project(device_protocol VERSION 2.2.0 LANGUAGES CXX)
add_library(mobocom_protocol STATIC mobocom_realtime/cpp/mobocom_protocol.cpp)
add_library(mobocom::protocol ALIAS mobocom_protocol)
target_compile_features(mobocom_protocol PUBLIC cxx_std_20)
target_include_directories(mobocom_protocol PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/mobocom_realtime/cpp>)
add_library(mobocom_timecode INTERFACE)
add_library(mobocom::timecode ALIAS mobocom_timecode)
target_compile_features(mobocom_timecode INTERFACE cxx_std_20)
target_include_directories(mobocom_timecode INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/timecode/cpp>)
include(CTest)
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(protocol_sdk_test mobocom_realtime/tests/test_sdk.cpp)
    target_link_libraries(protocol_sdk_test PRIVATE mobocom::protocol mobocom::timecode)
    add_test(NAME protocol_cpp COMMAND protocol_sdk_test)
    add_executable(protocol_configuration_test mobocom_realtime/tests/test_configuration.cpp)
    target_link_libraries(protocol_configuration_test PRIVATE mobocom::protocol)
    add_test(NAME protocol_configuration COMMAND protocol_configuration_test)
    add_executable(protocol_motion_profile_test mobocom_realtime/tests/test_motion_profile.cpp)
    target_link_libraries(protocol_motion_profile_test PRIVATE mobocom::protocol)
    add_test(NAME protocol_motion_profile COMMAND protocol_motion_profile_test)
    add_executable(protocol_focus_control_test mobocom_realtime/tests/test_focus_control.cpp)
    target_link_libraries(protocol_focus_control_test PRIVATE mobocom::protocol)
    add_test(NAME protocol_focus_control COMMAND protocol_focus_control_test)
    add_test(NAME protocol_reference COMMAND ${Python3_EXECUTABLE} -m unittest discover -s mobocom_realtime -t .)
    set_tests_properties(protocol_reference PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
endif()
